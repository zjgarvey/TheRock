cmake_minimum_required(VERSION 3.25)
project(therock-rocm-runtimes LANGUAGES C CXX)

set(ROCM_RUNTIMES_SOURCE_DIR "" CACHE PATH "Standalone rocm-systems runtimes source directory")
if(NOT IS_ABSOLUTE "${ROCM_RUNTIMES_SOURCE_DIR}" OR
   NOT EXISTS "${ROCM_RUNTIMES_SOURCE_DIR}/CMakeLists.txt" OR
   NOT EXISTS "${ROCM_RUNTIMES_SOURCE_DIR}/Cargo.lock" OR
   NOT EXISTS "${ROCM_RUNTIMES_SOURCE_DIR}/rust-toolchain.toml")
  message(FATAL_ERROR "ROCM_RUNTIMES_SOURCE_DIR must name a prepared standalone runtimes source tree")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The runtimes packaging adapter requires Linux ELF shared libraries")
endif()

option(THEROCK_BUILD_TESTING "Build the upstream runtimes host tests" OFF)
set(BUILD_TESTING "${THEROCK_BUILD_TESTING}" CACHE BOOL "Build upstream tests" FORCE)
include(CTest)

# The standalone runtimes build has no integrated CTS runner yet.
# Ignored GPU tests in Cargo's rocddi test binary are a separate interface.
option(ROCM_RUNTIMES_BUILD_CTS "Build the upstream CTS integration" OFF)
set(ROCM_RUNTIMES_CTS_AVAILABLE OFF CACHE INTERNAL "Upstream CTS availability" FORCE)
message(STATUS "Standalone runtimes CTS integration: unavailable")
if(ROCM_RUNTIMES_BUILD_CTS)
  message(FATAL_ERROR "CTS was requested, but this upstream runtimes build has no CTS integration")
endif()

set(_upstream_binary_dir "${CMAKE_CURRENT_BINARY_DIR}/upstream")
# Keep all installation under this adapter's ownership, including if upstream
# later adds install rules. Cargo, its frozen/offline policy, and the compiler
# environment are supplied by upstream's RuntimeRust.cmake.
add_subdirectory("${ROCM_RUNTIMES_SOURCE_DIR}" "${_upstream_binary_dir}" EXCLUDE_FROM_ALL)
foreach(_target runtime_shared_image runtime_amdf_static rocm_runtime_rust)
  if(NOT TARGET "${_target}")
    message(FATAL_ERROR "Standalone runtimes must provide ${_target}")
  endif()
endforeach()
add_custom_target(rocm-runtimes-build ALL DEPENDS rocm_runtime_rust)

if(THEROCK_BUILD_TESTING)
  # These are the native host executables registered by upstream's DDI build.
  foreach(_target amdf_smoke_shared amdf_smoke_static amdf_layout hsa_smoke)
    if(NOT TARGET "${_target}")
      message(FATAL_ERROR "Standalone runtimes is missing host test target ${_target}")
    endif()
    add_dependencies(rocm-runtimes-build "${_target}")
  endforeach()
  foreach(_target hsa_fork_smoke hsa_fork_tool)
    if(TARGET "${_target}")
      add_dependencies(rocm-runtimes-build "${_target}")
    endif()
  endforeach()
  add_custom_target(rocm-runtimes-host-tests
    COMMAND "${CMAKE_CTEST_COMMAND}" --test-dir "${CMAKE_CURRENT_BINARY_DIR}"
      --output-on-failure --no-tests=error -L host
      --output-junit "${CMAKE_CURRENT_BINARY_DIR}/rocm-runtimes-host-tests.xml"
    DEPENDS rocm-runtimes-build
    USES_TERMINAL
    VERBATIM
  )
endif()

# Reuse the pinned toolchain, prepared Cargo home/config, and native compiler
# environment generated by upstream. Formatting does not resolve dependencies.
include("${_upstream_binary_dir}/runtime-rust-config.cmake")
add_custom_target(rocm-runtimes-format-check
  COMMAND ${_runtime_cargo_command} fmt --all --check
  WORKING_DIRECTORY "${ROCM_RUNTIMES_SOURCE_DIR}"
  USES_TERMINAL
  VERBATIM
)
if(THEROCK_BUILD_TESTING)
  add_test(NAME runtimes.format COMMAND ${_runtime_cargo_command} fmt --all --check)
  set_tests_properties(runtimes.format PROPERTIES
    WORKING_DIRECTORY "${ROCM_RUNTIMES_SOURCE_DIR}"
    LABELS "host;format"
    RESOURCE_LOCK runtime_cargo
  )
endif()

get_target_property(_image runtime_shared_image IMPORTED_LOCATION)
get_target_property(HSA_SONAME runtime_shared_image IMPORTED_SONAME)
cmake_path(GET _image FILENAME HSA_IMAGE_NAME)
if(NOT HSA_IMAGE_NAME MATCHES "^libhsa-runtime64\\.so\\.(1\\.[0-9]+\\.[0-9]+)$")
  message(FATAL_ERROR "Expected a versioned SONAME 1 HSA image from standalone runtimes")
endif()
set(HSA_VERSION "${CMAKE_MATCH_1}")
if(NOT HSA_SONAME STREQUAL "libhsa-runtime64.so.1")
  message(FATAL_ERROR "Standalone runtimes must expose libhsa-runtime64.so.1")
endif()

set(_private_prefix "lib/rocm-runtimes")
install(FILES "$<TARGET_FILE:runtime_shared_image>" DESTINATION "${_private_prefix}/lib")
install(FILES "$<TARGET_FILE:runtime_amdf_static>" DESTINATION "${_private_prefix}/lib")
# This response file is part of the static target's upstream link contract;
# retain its ordered native libraries without exporting a build-tree target.
get_target_property(_native_response runtime_amdf_static INTERFACE_LINK_DEPENDS)
if(NOT _native_response)
  message(FATAL_ERROR "The static AMDF target must provide its native library response file")
endif()
install(FILES ${_native_response} DESTINATION "${_private_prefix}/lib")
set(HSA_LIBRARY_DESTINATION "${_private_prefix}/lib")
set(INSTALL_AMDF_ALIASES ON)
configure_file(install_symlinks.cmake.in install_symlinks.cmake @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/install_symlinks.cmake")
install(DIRECTORY "${ROCM_RUNTIMES_SOURCE_DIR}/api-headers/include/"
  DESTINATION "${_private_prefix}/include"
  FILES_MATCHING PATTERN "*.h"
  PATTERN "hsa" EXCLUDE
)
include(PublicHsaHeaders.cmake)
rocm_runtimes_prepare_hsa_headers(
  "${ROCM_RUNTIMES_SOURCE_DIR}/api-headers/include/hsa"
  "${CMAKE_CURRENT_BINARY_DIR}/public-hsa"
)
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/public-hsa/"
  DESTINATION "${_private_prefix}/include/hsa"
)
set(ROCM_RUNTIMES_HOST_TESTS_ENABLED false)
if(THEROCK_BUILD_TESTING)
  set(ROCM_RUNTIMES_HOST_TESTS_ENABLED true)
endif()
configure_file(runtime.json.in runtime.json @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/runtime.json"
  DESTINATION "${_private_prefix}/share/rocm-runtimes"
)
install(FILES LICENSE LICENSE-HSA.txt
  DESTINATION "${_private_prefix}/share/doc/rocm-runtimes"
)
set(ROCM_RUNTIMES_LICENSE_FILES "" CACHE STRING "Additional dependency and toolchain license/notice files")
set(_license_names LICENSE LICENSE-HSA.txt)
foreach(_license IN LISTS ROCM_RUNTIMES_LICENSE_FILES)
  if(NOT IS_ABSOLUTE "${_license}" OR NOT EXISTS "${_license}" OR IS_DIRECTORY "${_license}")
    message(FATAL_ERROR "ROCM_RUNTIMES_LICENSE_FILES must contain existing absolute file paths")
  endif()
  cmake_path(GET _license FILENAME _license_name)
  if(_license_name IN_LIST _license_names)
    message(FATAL_ERROR "Runtime license/notice files must have unique basenames")
  endif()
  list(APPEND _license_names "${_license_name}")
  install(FILES "${_license}" DESTINATION "${_private_prefix}/share/doc/rocm-runtimes")
endforeach()
